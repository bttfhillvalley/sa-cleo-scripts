{$CLEO .s}
{$INCLUDE ../Common/CleoConstants.txt}
0000:
int vehicle
int chaser
int wheel_state
int 32@
while true
    wait 0
    if
        Car.Wrecked(vehicle)
    then
        terminate_this_custom_script
    end
    // Reset lights
    3F10: set_car vehicle component "bottomlights" visible 0
    for chaser = 1 to 5
        3F11: set_car vehicle component "chaserlights" index chaser visible 0
    end
    if
        3F40:   is_car vehicle component "underbodybttf2" visible
    then
        3F27: get_car vehicle wheel_status_to wheel_state
        if
            not wheel_state == 0
        then
            // TODO: 3F52 is not executing
            // 3F52: fade_car vehicle component "bottomlights" alpha 255 step 15
            // Workaround:
            cleo_call @FadeBottomLights 3 vehicle 255 15
            if
                timera >= 1250
            then
                timera = 0
            end

            chaser = timera
            chaser /= 250
            chaser += 1

            3F11: set_car vehicle component "chaserlights" index chaser visible 1
        else
            // TODO: 3F52 is not executing
            // 3F52: fade_car vehicle component "bottomlights" alpha 0 step 15
            // Workaround:
            cleo_call @FadeBottomLights 3 vehicle 0 15
        end
    end
end

:FadeBottomLights
gosub @FadeLimit
3F47: 15@ = get_car 0@ component "bottomlights" alpha
gosub @Fade
3F12: set car 0@ component "bottomlights" alpha 15@
cleo_return 0

:FadeLimit
if
    1@ < 0
then
    1@ = 0
end
if
    1@ > 255
then
    1@ = 255
end
return

:Fade
if
    001D: 15@ > 1@
then
    0062: 15@ -= 2@
    if
        001D: 1@ > 15@
    then
        0085: 15@ = 1@
    end
else if
    001D: 1@ > 15@
then

    005A: 15@ += 2@
    if
        001D: 15@ > 1@
    then
        0085: 15@ = 1@
    end
end
end
return
