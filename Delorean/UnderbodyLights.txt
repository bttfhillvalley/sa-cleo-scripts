{$CLEO .s}
{$INCLUDE ../Common/CleoConstants.txt}
0000:
int vehicle
int chaser
int wheel_state
int 32@
while true
    wait 0
    if
        Car.Wrecked(vehicle)
    then
        terminate_this_custom_script
    end
    // Reset lights
    3F10: set_car vehicle component "bottomlights" visible 0
    for chaser = 1 to 5
        3F11: set_car vehicle component "chaserlights" index chaser visible 0
    end
    if
        3F40:   is_car vehicle component "underbodybttf2" visible
    then
        3F27: get_car vehicle wheel_status_to wheel_state
        if
            // TODO: use constant HOVER_WHEELS when opcode 3F27 is updated with individual wheel data
            not wheel_state == 0
        then
            3F52: fade_car vehicle component "bottomlights" alpha 255 step 15
            if
                timera >= 1250
            then
                timera = 0
            end

            chaser = timera
            chaser /= 250
            chaser += 1

            3F11: set_car vehicle component "chaserlights" index chaser visible 1
        else
            3F52: fade_car vehicle component "bottomlights" alpha 0 step 15
        end
    end
end
