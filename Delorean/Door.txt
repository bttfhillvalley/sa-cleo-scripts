{$CLEO .s}
{$INCLUDE ../Common/CleoConstants.txt}
0000:
int vehicle
int left_door = 1
int right_door = 1
int left
int right
float left_angle
float left_angle_sin
float right_angle
float right_angle_sin
float left_deg
float right_deg
float left_piston
float left_strut_angle
float right_piston
float right_strut_angle
int ignore
const
    BODY_PIVOT = 0.364084
    BODY_PIVOT_SQR = 0.132557
    DOOR_PIVOT = 0.105361
    DOOR_PIVOT_SQR = 0.011100
end
while true
    wait 0
    if
        Car.Wrecked(vehicle)
    then
        terminate_this_custom_script
    end
    095F: get_car vehicle door DOOR_FRONT_LEFT angle_to left_angle // Returns Rads
    095F: get_car vehicle door DOOR_FRONT_RIGHT angle_to right_angle // Returns Rads

    //Converting radians to degrees
    left_deg = left_angle 
    left_deg *= RAD_TO_DEG
    3F16: set_car vehicle component "fxdoorlf_"  angle 0 0 left_deg // Works with degrees
    
    left_piston = left_deg
    0097: abs left_piston
    02F7: left_piston = cosine left_piston
    left_piston *= BODY_PIVOT
    left_piston *= DOOR_PIVOT
    left_piston *= 2.0
    left_piston *= -1.0
    left_piston += DOOR_PIVOT_SQR
    left_piston += BODY_PIVOT_SQR
    01FB: left_piston = square_root left_piston

    02F6: left_angle_sin = sine left_deg
    0097: abs left_angle_sin
    left_strut_angle = DOOR_PIVOT
    left_strut_angle /= left_piston
    left_strut_angle *= left_angle_sin
    0AA5: call 0x4207A0 num_params 1 pop 1 left_strut_angle
    0AE9: pop_float left_strut_angle

    // To degrees
    left_strut_angle *= RAD_TO_DEG
    left_strut_angle *= -1.0
    left_strut_angle += 86.6

    3F16: set_car vehicle component "doorlfstrut_"  angle 0 left_strut_angle 0

    left_piston -= 0.259
    3F14: set_car vehicle component "doorlfstrutp"   pos  0 0 left_piston

    //Converting radians to degrees
    right_deg = right_angle
    right_deg *= RAD_TO_DEG
    3F16: set_car vehicle component "fxdoorrf_"  angle 0 0 right_deg
    
    right_piston = right_deg
    0097: abs right_piston
    02F7: right_piston = cosine right_piston
    right_piston *= BODY_PIVOT
    right_piston *= DOOR_PIVOT
    right_piston *= 2.0
    right_piston *= -1.0
    right_piston += DOOR_PIVOT_SQR
    right_piston += BODY_PIVOT_SQR
    01FB: right_piston = square_root right_piston

    02F6: right_angle_sin = sine right_deg
    0097: abs right_angle_sin
    right_strut_angle = DOOR_PIVOT
    right_strut_angle /= right_piston
    right_strut_angle *= right_angle_sin
    0AA5: call 0x4207A0 num_params 1 pop 1 right_strut_angle
    0AE9: pop_float right_strut_angle

    // To degrees
    right_strut_angle *= RAD_TO_DEG
    right_strut_angle *= -1.0
    right_strut_angle += 86.6
    right_strut_angle *= -1.0

    3F16: set_car vehicle component "doorrfstrut_"  angle 0 right_strut_angle 0 

    right_piston -= 0.259
    3F14: set_car vehicle component "doorrfstrutp"   pos  0 0 right_piston

    if
        left_angle == 0.0
    then
        left = 1
    else
        left = 0
    end
    if
        right_angle == 0.0
    then
        right = 1
    else
        right = 0
    end
    if and
        left_door == 1
        left == 0
    then
        3F96: ignore = attach_sound "delorean_door.wav" to_car vehicle offset -1.0 0 0 loop 0 min 10.0
    end
    if and
        right_door == 1
        right == 0
    then
        3F96: ignore = attach_sound "delorean_door.wav" to_car vehicle offset 1.0 0 0 loop 0 min 10.0
    end
    left_door = left
    right_door = right
end
